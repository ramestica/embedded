# embedded/CMakeLists.txt
cmake_minimum_required(VERSION 3.25)

project(
    embedded
    VERSION 1.0.0
    LANGUAGES C CXX
)

set(EMB_CONFIG_DIR "${CMAKE_BINARY_DIR}/config")

# Modules use this to guard sources/libraries that only make sense on one
# side (e.g. bare-metal syscall stubs vs. a hosted libc) instead of on
# CMAKE_CROSSCOMPILING directly.
if(CMAKE_CROSSCOMPILING)
    set(EMB_HOST FALSE)
else()
    set(EMB_HOST TRUE)
endif()

list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

include(helpers)
include(module)
include(library)
include(executable)
include(script)
include(config)
include(container)
include(uv)

include(GNUInstallDirs)
include(vendor)

enable_testing()

#
# Modules
#
# containers/devenv only stages Docker/script config files (cmake/config.cmake,
# cmake/script.cmake) and needs no toolchain. application/ pulls in Conan-
# resolved C/C++ dependencies (containers/devenv/config/conanfile.py) via the
# conan_toolchain.cmake generated by `make conan-install` -- skip it when all
# you need is the devenv container image.
option(EMB_BUILD_APPLICATION "Configure application/ (requires the Conan-generated toolchain)" ON)

add_subdirectory(containers/devenv)
if(EMB_BUILD_APPLICATION)
    add_subdirectory(application)
endif()
